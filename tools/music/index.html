hex_alpha <- function(hex, alpha) {
  alpha <- max(0, min(1, alpha))
  a <- toupper(sprintf("%02X", round(alpha * 255)))
  paste0(hex, a)
}

selected_effects <- function(structure_id, factor_id, category_id) {
  if (is.null(factor_id) || factor_id == "None" ||
      is.null(category_id) || category_id == "") return(NULL)

  regression[
    regression$structure == structure_id &
    regression$factor == factor_id &
    regression$category == category_id,
  ]
}

# ============================================================
# SVG NETWORK
# ============================================================

network_svg <- function(structure_id, effects = NULL) {

  ee <- edges[edges$structure == structure_id, ]
  max_abs_edge <- max(abs(ee$r))
  W <- 445
  H <- 390

  svg <- paste0(
    '<svg viewBox="0 0 ', W, ' ', H,
    '" style="width:100%;height:auto;display:block;">'
  )

  # CCA edges: always represent the relational taste structure.
  for (i in seq_len(nrow(ee))) {
    a <- node_pos[node_pos$code == ee$from[i], ]
    b <- node_pos[node_pos$code == ee$to[i], ]
    rr <- ee$r[i]
    mag <- abs(rr)

    opacity <- 0.12 + 0.78 * (mag / max_abs_edge)
    width <- 0.7 + 5.5 * (mag / max_abs_edge)
    col <- if (rr >= 0) edge_pos else edge_neg
    dash <- if (rr < 0) ' stroke-dasharray="7 5"' else ""

    svg <- paste0(
      svg,
      '<line x1="', a$x, '" y1="', a$y,
      '" x2="', b$x, '" y2="', b$y,
      '" stroke="', col,
      '" stroke-width="', sprintf("%.2f", width),
      '" opacity="', sprintf("%.2f", opacity), '"',
      dash,
      ' stroke-linecap="round"/>'
    )

    if (mag >= 0.20) {
      mx <- (a$x + b$x) / 2
      my <- (a$y + b$y) / 2
      svg <- paste0(
        svg,
        '<rect x="', sprintf("%.1f", mx - 19),
        '" y="', sprintf("%.1f", my - 10),
        '" width="38" height="17" rx="8.5" fill="#ffffff" opacity=".90"/>',
        '<text x="', sprintf("%.1f", mx),
        '" y="', sprintf("%.1f", my + 2),
        '" text-anchor="middle" font-size="9" font-weight="600" fill="#4b5563">',
        sprintf("%.2f", rr),
        '</text>'
      )
    }
  }

  # Nodes: neutral when no social overlay; coefficient-coded otherwise.
  for (i in seq_len(nrow(node_pos))) {
    p <- node_pos[i, ]
    lab <- strsplit(labels[p$code], "\n", fixed = TRUE)[[1]]

    fill <- "#FFFFFF"
    stroke <- "#66717C"
    stroke_w <- 1.4
    beta_txt <- ""
    beta_col <- "#59636E"

    if (!is.null(effects)) {
      e <- effects[effects$dimension == p$code, ]
      if (nrow(e) == 1) {
        sig <- nchar(e$stars) > 0
        mag <- min(abs(e$beta) / 0.80, 1)

        if (sig) {
          base <- if (e$beta >= 0) pos_col else neg_col
          fill <- hex_alpha(base, 0.18 + 0.62 * mag)
          stroke <- base
          stroke_w <- 2.2
          beta_col <- "#263238"
        } else {
          fill <- "#F3F4F5"
          stroke <- "#B6BCC2"
          stroke_w <- 1.2
          beta_col <- "#8A929A"
        }

        beta_txt <- paste0(
          ifelse(e$beta > 0, "+", ""),
          sprintf("%.2f", e$beta),
          e$stars
        )
      }
    }

    svg <- paste0(
      svg,
      '<circle cx="', p$x, '" cy="', p$y,
      '" r="38" fill="', fill,
      '" stroke="', stroke, '" stroke-width="', stroke_w, '"/>'
    )

    if (length(lab) == 1) {
      svg <- paste0(
        svg,
        '<text x="', p$x, '" y="', p$y - ifelse(beta_txt=="", -4, 2),
        '" text-anchor="middle" font-size="10.5" font-weight="650" fill="#263238">',
        lab[1], '</text>'
      )
    } else {
      svg <- paste0(
        svg,
        '<text x="', p$x, '" y="', p$y - ifelse(beta_txt=="", 2, 7),
        '" text-anchor="middle" font-size="9.8" font-weight="650" fill="#263238">',
        '<tspan x="', p$x, '" dy="0">', lab[1], '</tspan>',
        '<tspan x="', p$x, '" dy="11">', lab[2], '</tspan>',
        '</text>'
      )
    }

    if (beta_txt != "") {
      svg <- paste0(
        svg,
        '<text x="', p$x, '" y="', p$y + 23,
        '" text-anchor="middle" font-size="9.5" font-weight="700" fill="', beta_col, '">',
        beta_txt, '</text>'
      )
    }
  }

  # Edge legend
  svg <- paste0(
    svg,
    '<line x1="14" y1="367" x2="44" y2="367" stroke="', edge_pos, '" stroke-width="3"/>',
    '<text x="51" y="371" font-size="9.2" fill="#68707b">positive CCA association</text>',
    '<line x1="226" y1="367" x2="256" y2="367" stroke="', edge_neg, '" stroke-width="3" stroke-dasharray="7 5"/>',
    '<text x="263" y="371" font-size="9.2" fill="#68707b">negative CCA association</text>',
    '</svg>'
  )

  HTML(svg)
}

# ============================================================
# SOCIAL COMPOSITION PANEL
# ============================================================

profile_html <- function(structure_id) {
  d <- profile[profile$structure == structure_id, ]
  d$delta <- d$value - d$total
  rows <- character(0)

  for (i in seq_len(nrow(d))) {
    scale_max <- 15
    delta <- max(-scale_max, min(scale_max, d$delta[i]))
    bar_w <- abs(delta) / scale_max * 46
    left <- if (delta >= 0) 50 else 50 - bar_w
    col <- if (delta >= 0) pos_col else neg_col

    rows <- c(rows, paste0(
      '<div class="profile-row">',
        '<div class="profile-label">', d$metric[i], '</div>',
        '<div class="profile-track">',
          '<div class="profile-zero"></div>',
          '<div class="profile-delta" style="left:', sprintf("%.1f", left),
          '%;width:', sprintf("%.1f", bar_w), '%;background:', col, ';"></div>',
        '</div>',
        '<div class="profile-value">',
          sprintf("%.1f%%", d$value[i]),
          '<span class="profile-diff">',
          ifelse(d$delta[i] >= 0, "+", ""),
          sprintf("%.1f pp", d$delta[i]),
          '</span>',
        '</div>',
      '</div>'
    ))
  }

  HTML(paste0(
    paste(rows, collapse = ""),
    '<div class="profile-foot">Percentage-point difference from the full sample.</div>'
  ))
}

# ============================================================

taste_profile_svg <- function(structure_id) {
  d <- taste_scores[taste_scores$structure == structure_id, ]
  d <- d[match(c("MU","PR","JB","F","OC","A"), d$dimension), ]

  W <- 430
  H <- 178
  left <- 142
  right <- 28
  top <- 10
  row_h <- 27
  zero_x <- left + (W-left-right) * (0 - (-0.8)) / (1.3 - (-0.8))

  xmap <- function(v) left + (W-left-right) * (v - (-0.8)) / (1.3 - (-0.8))

  svg <- paste0('<svg viewBox="0 0 ',W,' ',H,'" style="width:100%;height:auto;display:block;">')
  svg <- paste0(svg,
    '<line x1="',zero_x,'" y1="6" x2="',zero_x,'" y2="166" stroke="#AEB4BA" stroke-width="1"/>'
  )

  for (i in seq_len(nrow(d))) {
    y <- top + (i-1)*row_h + 11
    v <- d$score[i]
    x2 <- xmap(v)
    x1 <- zero_x
    col <- if (v >= 0) pos_col else neg_col

    svg <- paste0(
      svg,
      '<text x="2" y="',y+3,'" font-size="9.4" fill="#46515C">',
      labels[d$dimension[i]], '</text>',
      '<line x1="',min(x1,x2),'" y1="',y,'" x2="',max(x1,x2),
      '" y2="',y,'" stroke="',col,'" stroke-width="7" stroke-linecap="round"/>',
      '<circle cx="',x2,'" cy="',y,'" r="4.2" fill="',col,'"/>',
      '<text x="', ifelse(v>=0, x2+8, x2-8), '" y="',y+3,
      '" text-anchor="', ifelse(v>=0,"start","end"),
      '" font-size="9.4" font-weight="650" fill="#37414B">',
      sprintf("%.2f",v), '</text>'
    )
  }

  svg <- paste0(svg, '</svg>')
  HTML(svg)
}

# ============================================================
# SERVER
# ============================================================

server <- function(input, output, session) {

  output$category_ui <- renderUI({
    if (is.null(input$factor) || input$factor == "None") {
      selectInput("category", NULL, choices=c("No overlay"=""), selected="", disabled=TRUE)
    } else {
      selectInput(
        "category", NULL,
        choices=category_choices[[input$factor]],
        selected=category_choices[[input$factor]][1]
      )
    }
  })

  selected_info <- reactive({
    structures[structures$id == input$structure, ]
  })

  effects <- reactive({
    selected_effects(input$structure, input$factor, input$category)
  })

  output$structure_header <- renderUI({
    s <- selected_info()
    div(
      div(class="structure-title", s$name),
      div(
        class="structure-meta",
        paste0(s$id, " · ", s$share, "% of respondents · n = ", format(s$n, big.mark=","))
      )
    )
  })

  output$overlay_info <- renderUI({
    e <- effects()

    if (is.null(e)) {
      return(
        div(
          class="overlay-info",
          HTML("<strong>No social overlay selected.</strong> Nodes are neutral; the network shows only the CCA relational structure.")
        )
      )
    }

    ref <- unique(e$reference)
    div(
      class="overlay-info",
      HTML(paste0(
        "<strong>", input$factor, " · ", input$category, "</strong> ",
        "— node coefficients are relative to the reference category <strong>", ref, "</strong>."
      ))
    )
  })

  output$network <- renderUI({
    network_svg(input$structure, effects())
  })

  output$profile <- renderUI({
    profile_html(input$structure)
  })

  output$taste_profile <- renderUI({
    taste_profile_svg(input$structure)
  })
}

shinyApp(ui, server)
