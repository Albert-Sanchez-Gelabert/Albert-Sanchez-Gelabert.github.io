prob <- data.frame(
  concentration = rep(c("Low", "Medium", "High"), each = 4),
  gender = rep(rep(c("Female", "Male"), each = 2), 3),
  background = rep(c("Native", "Non-native"), 6),

  dropout = c(
    .051, .129, .042, .112,
    .073, .149, .058, .117,
    .080, .139, .088, .140
  ),

  vocational = c(
    .237, .298, .245, .328,
    .276, .320, .306, .376,
    .316, .321, .347, .384
  ),

  academic = c(
    .712, .572, .713, .560,
    .651, .531, .636, .507,
    .603, .540, .565, .476
  ),

  stringsAsFactors = FALSE
)

context_meta <- data.frame(
  concentration = c("Low", "Medium", "High"),
  range = c("0–15%", "16–30%", "31%+"),
  stringsAsFactors = FALSE
)

outcome_meta <- list(
  Academic = list(
    key = "academic",
    min = .45,
    max = .75,
    colour = "#5B8E7D"
  ),
  Vocational = list(
    key = "vocational",
    min = .20,
    max = .40,
    colour = "#D9823F"
  ),
  Dropout = list(
    key = "dropout",
    min = .03,
    max = .16,
    colour = "#527CA5"
  )
)

# ============================================================
# SVG: two gender rows, Native vs Non-native on each row
# ============================================================

panel_svg <- function(d, outcome_name) {

  meta <- outcome_meta[[outcome_name]]

  W <- 320
  H <- 205

  x0 <- 78
  x1 <- 278

  y_f <- 72
  y_m <- 145

  xmap <- function(v) {
    x0 + (v - meta$min) / (meta$max - meta$min) * (x1 - x0)
  }

  getv <- function(g, b) {
    d[d$gender == g & d$background == b, meta$key]
  }

  fn <- getv("Female", "Native")
  fz <- getv("Female", "Non-native")
  mn <- getv("Male", "Native")
  mz <- getv("Male", "Non-native")

  x_fn <- xmap(fn)
  x_fz <- xmap(fz)
  x_mn <- xmap(mn)
  x_mz <- xmap(mz)

  gap_f <- 100 * (fn - fz)
  gap_m <- 100 * (mn - mz)

  row_svg <- function(y, label, xn, xz, vn, vz, gap) {

    paste0(
      # gender label
      '<text x="8" y="', y+4,
      '" font-size="11" font-weight="700" fill="#444">',
      label, '</text>',

      # row scale
      '<line x1="', x0, '" y1="', y,
      '" x2="', x1, '" y2="', y,
      '" stroke="#dddddd" stroke-width="2" stroke-linecap="round"/>',

      # connector
      '<line x1="', min(xn,xz), '" y1="', y,
      '" x2="', max(xn,xz), '" y2="', y,
      '" stroke="', meta$colour,
      '" stroke-width="5" stroke-linecap="round" opacity=".32"/>',

      # Native = solid
      '<circle cx="', xn, '" cy="', y,
      '" r="7.5" fill="', meta$colour,
      '" stroke="#fff" stroke-width="1.8"/>',

      # Non-native = open
      '<circle cx="', xz, '" cy="', y,
      '" r="7" fill="#fff" stroke="', meta$colour,
      '" stroke-width="2.5"/>',

      # Values
      '<text x="', xn, '" y="', y-15,
      '" text-anchor="middle" font-size="10.5" font-weight="700" fill="#333">',
      sprintf("%.1f%%",100*vn), '</text>',

      '<text x="', xz, '" y="', y+24,
      '" text-anchor="middle" font-size="10.5" font-weight="700" fill="#555">',
      sprintf("%.1f%%",100*vz), '</text>',

      # gap on far right
      '<text x="310" y="', y+4,
      '" text-anchor="end" font-size="9" font-weight="700" fill="#777">',
      'Δ ', sprintf("%.1f",gap), '</text>'
    )
  }

  HTML(
    paste0(
      '<svg viewBox="0 0 ',W,' ',H,
      '" style="width:100%;height:auto;display:block;">',

      # top labels / legend integrated
      '<circle cx="83" cy="18" r="5" fill="', meta$colour, '"/>',
      '<text x="94" y="22" font-size="9.5" fill="#666">Native</text>',

      '<circle cx="145" cy="18" r="5" fill="#fff" stroke="', meta$colour, '" stroke-width="2"/>',
      '<text x="156" y="22" font-size="9.5" fill="#666">Non-native</text>',

      '<text x="310" y="22" text-anchor="end" font-size="8.5" fill="#888">Δ Native − Non-native</text>',

      # rows
      row_svg(y_f, "Female", x_fn, x_fz, fn, fz, gap_f),
      row_svg(y_m, "Male", x_mn, x_mz, mn, mz, gap_m),

      # faint separator
      '<line x1="8" y1="108" x2="310" y2="108" stroke="#f0f0f0" stroke-width="1"/>',

      # shared local scale
      '<text x="',x0,'" y="197" text-anchor="start" font-size="8.5" fill="#999">',
      sprintf("%.0f%%",100*meta$min), '</text>',
      '<text x="',x1,'" y="197" text-anchor="end" font-size="8.5" fill="#999">',
      sprintf("%.0f%%",100*meta$max), '</text>',

      '</svg>'
    )
  )
}

# ============================================================
# SERVER
# ============================================================

server <- function(input, output, session) {

  output$outcome_title <- renderUI({
    div(
      class = "outcome-title",
      paste0(input$outcome, " transition")
    )
  })

  output$contexts <- renderUI({

    panels <- lapply(
      c("Low","Medium","High"),
      function(conc) {

        d <- prob[
          prob$concentration == conc,
        ]

        cm <- context_meta[
          context_meta$concentration == conc,
        ]

        div(
          class = "context-card",

          div(
            class = "context-head",
            div(
              class = "context-name",
              paste0(conc, " concentration")
            ),
            div(
              class = "context-range",
              paste0(cm$range, " non-native students")
            )
          ),

          div(
            class = "context-body",
            panel_svg(
              d,
              input$outcome
            )
          )
        )
      }
    )

    div(class = "contexts", panels)
  })

  output$reading <- renderUI({

    meta <- outcome_meta[[input$outcome]]

    get_gap <- function(conc, gender) {
      d <- prob[
        prob$concentration == conc &
        prob$gender == gender,
      ]
      100 * (
        d[d$background == "Native", meta$key] -
        d[d$background == "Non-native", meta$key]
      )
    }

    low_f <- get_gap("Low","Female")
    high_f <- get_gap("High","Female")
    low_m <- get_gap("Low","Male")
    high_m <- get_gap("High","Male")

    div(
      class = "reading",

      div(
        class = "reading-label",
        "Migrant-background gap"
      ),

      div(
        class = "reading-text",
        paste0(
          "For women, the Native–Non-native gap changes from ",
          sprintf("%.1f", abs(low_f)), " pp at low concentration to ",
          sprintf("%.1f", abs(high_f)), " pp at high concentration. ",
          "For men, it changes from ",
          sprintf("%.1f", abs(low_m)), " to ",
          sprintf("%.1f", abs(high_m)), " pp."
        )
      )
    )
  })
}

shinyApp(ui, server)
