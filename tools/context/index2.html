library(shiny)

prob <- data.frame(
  concentration = rep(c("Low", "Medium", "High"), each = 4),
  gender = rep(rep(c("Female", "Male"), each = 2), 3),
  background = rep(c("Native", "Non-native"), 6),
  dropout = c(.051,.129,.042,.112,.073,.149,.058,.117,.080,.139,.088,.140),
  vocational = c(.237,.298,.245,.328,.276,.320,.306,.376,.316,.321,.347,.384),
  academic = c(.712,.572,.713,.560,.651,.531,.636,.507,.603,.540,.565,.476),
  stringsAsFactors = FALSE
)

context_meta <- data.frame(
  concentration = c("Low","Medium","High"),
  range = c("0–15%","16–30%","31%+"),
  stringsAsFactors = FALSE
)

outcome_meta <- list(
  Academic = list(key="academic", min=.45, max=.75, colour="#7F7F7F"),
  Vocational = list(key="vocational", min=.20, max=.40, colour="#D9823F"),
  Dropout = list(key="dropout", min=.03, max=.16, colour="#527CA5")
)

dumbbell_svg <- function(value_a, value_b, name_a, name_b, outcome_name) {
  meta <- outcome_meta[[outcome_name]]
  W <- 300; H <- 155; x0 <- 32; x1 <- 268; y <- 78
  xmap <- function(v) x0 + (v-meta$min)/(meta$max-meta$min)*(x1-x0)
  xa <- xmap(value_a); xb <- xmap(value_b)
  gap_abs <- abs(100*(value_a-value_b))
  close <- abs(xa-xb) < 42
  ya_label <- if (close) 39 else 43
  yb_label <- if (close) 122 else 113

  HTML(paste0(
    '<svg viewBox="0 0 ',W,' ',H,'" style="width:100%;height:auto;display:block;">',
    '<line x1="',x0,'" y1="',y,'" x2="',x1,'" y2="',y,'" stroke="#dddddd" stroke-width="2" stroke-linecap="round"/>',
    '<text x="',x0,'" y="',y+21,'" text-anchor="start" font-size="9" fill="#999">',sprintf("%.0f%%",100*meta$min),'</text>',
    '<text x="',x1,'" y="',y+21,'" text-anchor="end" font-size="9" fill="#999">',sprintf("%.0f%%",100*meta$max),'</text>',
    '<line x1="',min(xa,xb),'" y1="',y,'" x2="',max(xa,xb),'" y2="',y,'" stroke="',meta$colour,'" stroke-width="5" stroke-linecap="round" opacity=".35"/>',
    '<circle cx="',xa,'" cy="',y,'" r="8" fill="',meta$colour,'" stroke="#fff" stroke-width="2"/>',
    '<circle cx="',xb,'" cy="',y,'" r="7.5" fill="#fff" stroke="',meta$colour,'" stroke-width="3"/>',
    '<text x="',xa,'" y="',ya_label,'" text-anchor="middle" font-size="10" fill="#666">',name_a,'</text>',
    '<text x="',xa,'" y="',ya_label+15,'" text-anchor="middle" font-size="15" font-weight="700" fill="#333">',sprintf("%.1f%%",100*value_a),'</text>',
    '<text x="',xb,'" y="',yb_label,'" text-anchor="middle" font-size="10" fill="#666">',name_b,'</text>',
    '<text x="',xb,'" y="',yb_label+15,'" text-anchor="middle" font-size="15" font-weight="700" fill="#333">',sprintf("%.1f%%",100*value_b),'</text>',
    '<rect x="111" y="4" width="78" height="23" rx="11.5" fill="#fafafa" stroke="#e4e4e4"/>',
    '<text x="150" y="19" text-anchor="middle" font-size="10" font-weight="700" fill="#666">gap ',sprintf("%.1f",gap_abs),' pp</text>',
    '</svg>'
  ))
}

server <- function(input, output, session) {

  output$fixed_control <- renderUI({
    if (input$compare == "background") {
      radioButtons("fixed_gender","Gender",choices=c("Female","Male"),selected="Female",inline=TRUE)
    } else {
      radioButtons("fixed_background","Migrant background",choices=c("Native","Non-native"),selected="Native",inline=TRUE)
    }
  })

  comparison <- reactive({
    if (input$compare == "background") {
      req(input$fixed_gender)
      a <- prob[prob$gender == input$fixed_gender & prob$background == "Native",]
      b <- prob[prob$gender == input$fixed_gender & prob$background == "Non-native",]
      list(a=a,b=b,name_a="Native",name_b="Non-native")
    } else {
      req(input$fixed_background)
      a <- prob[prob$background == input$fixed_background & prob$gender == "Female",]
      b <- prob[prob$background == input$fixed_background & prob$gender == "Male",]
      list(a=a,b=b,name_a="Female",name_b="Male")
    }
  })

  output$outcome_title <- renderUI(div(class="selected-outcome",paste0(input$outcome," transition")))

  output$legend <- renderUI({
    x <- comparison()
    div(class="legend",
      div(class="legend-item",div(class="legend-solid"),x$name_a),
      div(class="legend-item",div(class="legend-open"),x$name_b)
    )
  })

  output$contexts <- renderUI({
    x <- comparison()
    meta <- outcome_meta[[input$outcome]]
    panels <- lapply(c("Low","Medium","High"), function(conc) {
      a <- x$a[x$a$concentration == conc,]
      b <- x$b[x$b$concentration == conc,]
      cm <- context_meta[context_meta$concentration == conc,]
      div(class="context-card",
        div(class="context-head",
          div(class="context-name",paste0(conc," concentration")),
          div(class="context-range",paste0(cm$range," non-native students"))
        ),
        div(class="context-body",
          dumbbell_svg(a[[meta$key]],b[[meta$key]],x$name_a,x$name_b,input$outcome)
        )
      )
    })
    div(class="contexts",panels)
  })

  output$gap_strip <- renderUI({
    x <- comparison()
    meta <- outcome_meta[[input$outcome]]
    ord <- match(c("Low","Medium","High"),x$a$concentration)
    a <- x$a[ord,]; b <- x$b[ord,]
    gaps <- abs(100*(a[[meta$key]]-b[[meta$key]]))

    direction <- if (gaps[3] < gaps[1]) {
      paste0("The gap narrows from ",sprintf("%.1f",gaps[1])," pp in low-concentration schools to ",sprintf("%.1f",gaps[3])," pp in high-concentration schools.")
    } else if (gaps[3] > gaps[1]) {
      paste0("The gap widens from ",sprintf("%.1f",gaps[1])," pp in low-concentration schools to ",sprintf("%.1f",gaps[3])," pp in high-concentration schools.")
    } else {
      paste0("The gap is ",sprintf("%.1f",gaps[1])," pp in both the low- and high-concentration scenarios.")
    }

    div(class="gap-strip",
      div(class="gap-strip-title","Across contexts"),
      div(class="gap-strip-copy",direction)
    )
  })
}

shinyApp(ui, server)
