esc <- function(x) {
  as.character(htmltools::htmlEscape(x))
}

svg_text <- function(x, y, text, size = 12, colour = "#555555",
                     anchor = "start", weight = 400, class = NULL) {
  cls <- if (is.null(class)) "" else paste0(' class="', class, '"')
  paste0(
    '<text', cls,
    ' x="', x,
    '" y="', y,
    '" text-anchor="', anchor,
    '" font-size="', size,
    '" font-weight="', weight,
    '" fill="', colour,
    '" style="paint-order:stroke;stroke:white;stroke-width:4px;stroke-linejoin:round;">',
    esc(text),
    '</text>'
  )
}

# ============================================================
# SERVER
# ============================================================

server <- function(input, output, session) {

  observeEvent(input$clear_subfields, {
    updateSelectizeInput(
      session,
      "subfields",
      selected = character(0)
    )
  })

  output$space <- renderUI({

    layers <- input$layers
    if (is.null(layers)) {
      layers <- character(0)
    }

    # --------------------------------------------------------
    # Visible data
    # --------------------------------------------------------

    active_d <- coords[
      coords$type == "active" &
        coords$variable == input$variable,
      ,
      drop = FALSE
    ]

    sub_d <- coords[
      coords$type == "subfield",
      ,
      drop = FALSE
    ]

    if (!is.null(input$field_filter) && input$field_filter != "All") {
      sub_d <- sub_d[
        sub_d$stem == input$field_filter,
        ,
        drop = FALSE
      ]
    }

    selected <- input$subfields
    if (is.null(selected)) {
      selected <- character(0)
    }

    # --------------------------------------------------------
    # Fixed plotting range
    #
    # Keep the same factorial-space reference frame for every
    # variable, layer combination and subdiscipline filter.
    # This prevents the plot from rescaling during exploration
    # and makes positions directly comparable across views.
    # --------------------------------------------------------

    xr <- c(-0.80, 2.25)
    yr <- c(-2.20, 0.80)

    # --------------------------------------------------------
    # SVG geometry
    # --------------------------------------------------------

    W <- 1000
    H <- 570
    left <- 90
    right <- 970
    top <- 35
    bottom <- 495

    sx <- function(x) {
      left + (x - xr[1]) / (xr[2] - xr[1]) * (right - left)
    }

    sy <- function(y) {
      bottom - (y - yr[1]) / (yr[2] - yr[1]) * (bottom - top)
    }

    # --------------------------------------------------------
    # Axes
    # --------------------------------------------------------

    axis_html <- ""

    if (xr[1] <= 0 && xr[2] >= 0) {
      axis_html <- paste0(
        axis_html,
        '<line x1="', sx(0), '" y1="', top,
        '" x2="', sx(0), '" y2="', bottom,
        '" stroke="#cccccc" stroke-width="1"/>'
      )
    }

    if (yr[1] <= 0 && yr[2] >= 0) {
      axis_html <- paste0(
        axis_html,
        '<line x1="', left, '" y1="', sy(0),
        '" x2="', right, '" y2="', sy(0),
        '" stroke="#cccccc" stroke-width="1"/>'
      )
    }

    # --------------------------------------------------------
    # Clusters
    # --------------------------------------------------------

    cluster_html <- ""

    if ("clusters" %in% layers) {

      cluster_html <- paste(
        vapply(
          seq_len(nrow(clusters)),
          function(i) {

            px <- sx(clusters$dim1[i])
            py <- sy(clusters$dim2[i])

            radius <- 8 + clusters$share[i] * 0.45
            colour <- unname(cluster_colours[clusters$gender[i]])

            dx <- if (clusters$dim1[i] < 0) -12 else 12
            anchor <- if (clusters$dim1[i] < 0) "end" else "start"

            # Extra vertical separation for the very close C2/C3 pair
            dy <- if (clusters$cluster[i] == "C2") -13 else if (clusters$cluster[i] == "C3") 18 else -11

            title_txt <- paste0(
              clusters$cluster[i],
              " · ", sprintf("%.1f%%", clusters$share[i]),
              " · ", clusters$gender[i],
              " · ", clusters$orientation[i],
              " | ", clusters$profile[i],
              " | Dim 1: ", sprintf("%.3f", clusters$dim1[i]),
              " | Dim 2: ", sprintf("%.3f", clusters$dim2[i])
            )

            hover_txt <- paste0(
              clusters$cluster[i],
              " · ", sprintf("%.1f%%", clusters$share[i]),
              " · ", clusters$gender[i],
              " · ", clusters$orientation[i]
            )

            paste0(
              '<g class="cluster-point">',
              '<circle class="cluster-symbol"',
              ' cx="', px, '" cy="', py, '" r="', radius, '"',
              ' fill="', colour, '" fill-opacity="0.16"',
              ' stroke="', colour, '" stroke-width="2.3">',
              '<title>', esc(title_txt), '</title>',
              '</circle>',
              svg_text(
                px + dx,
                py + dy,
                clusters$cluster[i],
                size = 12,
                colour = colour,
                anchor = anchor,
                weight = 700
              ),
              svg_text(
                px + dx,
                py + dy + 14,
                hover_txt,
                size = 10,
                colour = colour,
                anchor = anchor,
                weight = 500,
                class = "hover-label"
              ),
              '</g>'
            )
          },
          character(1)
        ),
        collapse = ""
      )
    }

    # --------------------------------------------------------
    # Subdisciplines
    # --------------------------------------------------------

    subfield_html <- ""

    if ("subfields" %in% layers && nrow(sub_d) > 0) {

      subfield_html <- paste(
        vapply(
          seq_len(nrow(sub_d)),
          function(i) {

            px <- sx(sub_d$dim1[i])
            py <- sy(sub_d$dim2[i])

            is_selected <- sub_d$label[i] %in% selected
            fill <- if (is_selected) "#C95D5D" else "#bdbdbd"
            size <- if (is_selected) 8 else 5.5

            if (sub_d$stem[i] == "STEM") {
              symbol <- paste0(
                '<circle class="subfield-symbol"',
                ' cx="', px, '" cy="', py, '" r="', size, '"',
                ' fill="', fill, '" stroke="white" stroke-width="1.3"/>'
              )
            } else {
              symbol <- paste0(
                '<rect class="subfield-symbol"',
                ' x="', px - size, '" y="', py - size, '"',
                ' width="', 2 * size, '" height="', 2 * size, '"',
                ' rx="1.2" fill="', fill, '"',
                ' stroke="white" stroke-width="1.3"/>'
              )
            }

            dx <- if (sub_d$dim1[i] > 0.6) -10 else 10
            anchor <- if (sub_d$dim1[i] > 0.6) "end" else "start"

            title_txt <- paste0(
              sub_d$label[i],
              " · ", sub_d$stem[i],
              " | Dim 1: ", sprintf("%.3f", sub_d$dim1[i]),
              " | Dim 2: ", sprintf("%.3f", sub_d$dim2[i])
            )

            persistent <- ""
            if (is_selected && isTRUE(input$labels)) {
              persistent <- svg_text(
                px + dx,
                py - 9,
                sub_d$label[i],
                size = 11,
                colour = "#333333",
                anchor = anchor,
                weight = 600
              )
            }

            paste0(
              '<g class="subfield-point">',
              symbol,
              '<title>', esc(title_txt), '</title>',
              svg_text(
                px + dx,
                py - 9,
                sub_d$label[i],
                size = 11,
                colour = "#666666",
                anchor = anchor,
                weight = 500,
                class = "hover-label"
              ),
              '</g>',
              persistent
            )
          },
          character(1)
        ),
        collapse = ""
      )
    }

    # --------------------------------------------------------
    # Active variable
    # --------------------------------------------------------

    variable_html <- ""

    if ("variables" %in% layers && nrow(active_d) > 0) {

      variable_html <- paste(
        vapply(
          seq_len(nrow(active_d)),
          function(i) {

            px <- sx(active_d$dim1[i])
            py <- sy(active_d$dim2[i])

            dx <- if (active_d$dim1[i] > mean(xr)) -10 else 10
            anchor <- if (active_d$dim1[i] > mean(xr)) "end" else "start"

            title_txt <- paste0(
              active_d$label[i],
              " | Dim 1: ", sprintf("%.3f", active_d$dim1[i]),
              " | Dim 2: ", sprintf("%.3f", active_d$dim2[i])
            )

            label_html <- ""
            if (isTRUE(input$labels)) {
              label_html <- svg_text(
                px + dx,
                py - 9,
                active_d$label[i],
                size = 12,
                colour = "#333333",
                anchor = anchor,
                weight = 500
              )
            }

            paste0(
              '<circle cx="', px, '" cy="', py, '" r="7"',
              ' fill="#4E79A7" stroke="white" stroke-width="1.5">',
              '<title>', esc(title_txt), '</title>',
              '</circle>',
              label_html
            )
          },
          character(1)
        ),
        collapse = ""
      )
    }

    # --------------------------------------------------------
    # SVG
    # --------------------------------------------------------

    html <- paste0(
      '<svg viewBox="0 0 ', W, ' ', H, '"',
      ' preserveAspectRatio="xMidYMid meet"',
      ' style="width:100%;height:auto;display:block;">',

      '<rect x="', left, '" y="', top, '"',
      ' width="', right - left, '" height="', bottom - top, '"',
      ' fill="#ffffff" stroke="#eeeeee"/>',

      axis_html,
      cluster_html,
      subfield_html,
      variable_html,

      '<text x="', left, '" y="525" text-anchor="start"',
      ' font-size="11" fill="#777777">← Academic / linear</text>',

      '<text x="', right, '" y="525" text-anchor="end"',
      ' font-size="11" fill="#777777">Vocational / delayed →</text>',

      '<text x="530" y="552" text-anchor="middle"',
      ' font-size="12" font-weight="600" fill="#555555">',
      'Axis 1 — Life-course position and institutionalised access pathway',
      '</text>',

      '<text x="25" y="270" transform="rotate(-90 25 270)"',
      ' text-anchor="middle" font-size="12" font-weight="600" fill="#555555">',
      'Axis 2 — Academic and institutional constraint gradient',
      '</text>',

      '</svg>'
    )

    HTML(html)
  })

  output$selection_note <- renderUI({

    selected <- input$subfields

    if (is.null(selected) || length(selected) == 0) {
      return(NULL)
    }

    d <- coords[
      coords$type == "subfield" &
        coords$label %in% selected,
      ,
      drop = FALSE
    ]

    if (nrow(d) == 0) {
      return(NULL)
    }

    lines <- vapply(
      seq_len(nrow(d)),
      function(i) {
        paste0(
          "<strong>", esc(d$label[i]), "</strong>",
          " · ", d$stem[i],
          " · Dim 1 = ", sprintf("%.3f", d$dim1[i]),
          " · Dim 2 = ", sprintf("%.3f", d$dim2[i])
        )
      },
      character(1)
    )

    div(
      class = "selection-note",
      HTML(paste(lines, collapse = "<br>"))
    )
  })
}

shinyApp(
  ui = ui,
  server = server
)
