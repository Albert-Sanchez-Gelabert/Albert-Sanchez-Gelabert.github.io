state_legend <- function(key, label) {
  div(
    class = "legend-item",
    span(
      class = "legend-swatch",
      style = paste0("background:", state_col[key], ";")
    ),
    label
  )
}

cohort_svg <- function(d) {

  d <- d[order(d$year), ]

  W <- 430
  H <- 235

  left <- 42
  right <- 18
  top <- 18
  bottom <- 40

  ph <- H - top - bottom
  bar_gap <- 5
  n <- nrow(d)
  usable_w <- W - left - right
  bar_w <- (usable_w - bar_gap * (n - 1)) / n

  ymap <- function(v) top + (100 - v) / 100 * ph

  svg <- paste0(
    '<svg viewBox="0 0 ',W,' ',H,
    '" style="width:100%;height:auto;display:block;">'
  )

  # horizontal grid and y-axis labels
  for (p in c(0,25,50,75,100)) {
    yy <- ymap(p)
    svg <- paste0(
      svg,
      '<line x1="',left,'" y1="',yy,
      '" x2="',W-right,'" y2="',yy,
      '" stroke="#ededed" stroke-width="1"/>',
      '<text x="',left-8,'" y="',yy+3,
      '" text-anchor="end" font-size="8.5" fill="#8d8d8d">',
      p, '%</text>'
    )
  }

  # TraMineR-style state distribution bars
  for (i in seq_len(n)) {

    x <- left + (i - 1) * (bar_w + bar_gap)

    vals <- c(
      university = d$university[i],
      outside = d$outside[i],
      graduated = d$graduated[i]
    )

    lower <- 0

    for (k in c("university","outside","graduated")) {

      v <- vals[k]
      y_top <- ymap(lower + v)
      y_bottom <- ymap(lower)
      h <- y_bottom - y_top

      svg <- paste0(
        svg,
        '<rect x="',sprintf("%.2f",x),
        '" y="',sprintf("%.2f",y_top),
        '" width="',sprintf("%.2f",bar_w),
        '" height="',sprintf("%.2f",h),
        '" fill="',state_col[k],
        '" stroke="#555" stroke-width=".75"/>'
      )

      # Add percentage labels only when the segment is large enough
      # to remain legible inside the stacked bar.
      if (v >= 5) {

        label_y <- y_top + h / 2
        label_col <- "#FFFFFF"

        svg <- paste0(
          svg,
          '<text x="', sprintf("%.2f", x + bar_w / 2),
          '" y="', sprintf("%.2f", label_y + 3),
          '" text-anchor="middle" font-size="8.3" font-weight="650" fill="', label_col, '">',
          sprintf("%.1f%%", v),
          '</text>'
        )
      }

      lower <- lower + v
    }

    svg <- paste0(
      svg,
      '<text x="',sprintf("%.2f",x + bar_w/2),
      '" y="',H-14,
      '" text-anchor="middle" font-size="8.8" fill="#666">',
      year_labels[i], '</text>'
    )
  }

  svg <- paste0(
    svg,
    '<line x1="',left,'" y1="',top,
    '" x2="',left,'" y2="',top+ph,
    '" stroke="#666" stroke-width="1"/>',
    '<line x1="',left,'" y1="',top+ph,
    '" x2="',W-right,'" y2="',top+ph,
    '" stroke="#666" stroke-width="1"/>',
    '</svg>'
  )

  HTML(svg)
}

cohort_card <- function(d, category) {

  final <- d[d$year == max(d$year), ]

  div(
    class = "group-card",

    div(
      class = "group-head",
      div(
        class = "group-title",
        category
      ),
      div(
        class = "group-n",
        paste0("n = ", format(final$n, big.mark=","))
      )
    ),

    div(
      class = "plot-wrap",
      cohort_svg(d)
    ),

    div(
      class = "final-row",

      div(
        class = "final-item",
        span(class="final-dot", style=paste0("background:",state_col["university"],";")),
        span(class="final-label","University"),
        strong(sprintf("%.1f%%",final$university))
      ),

      div(
        class = "final-item",
        span(class="final-dot", style=paste0("background:",state_col["outside"],";")),
        span(class="final-label","Outside"),

      div(
        class = "final-item",

        span(class="final-label","Graduated"),
        strong(sprintf("%.1f%%",final$graduated))
      )
    )
  )
}

metrics_svg <- function(m) {

  cats <- category_order[[unique(m$variable)]]
  m <- m[match(cats, m$category), ]

  metrics <- data.frame(
    key = c("transfer","outside","returned","complex","graduated"),
    label = c(
      "Ever transferred",
      "Ever outside university",
      "Returned after an exit",
      "2+ state changes",
      "Graduated at final observation"
    ),
    stringsAsFactors = FALSE
  )

  W <- 860
  H <- 355
  left <- 215
  right <- 55
  top <- 42
  row_h <- 57
  max_x <- 65

  xmap <- function(v) {
    left + (v/max_x) * (W-left-right)
  }

  svg <- paste0(
    '<svg viewBox="0 0 ',W,' ',H,
    '" style="width:100%;height:auto;display:block;">'
  )

  # axis
  for (p in c(0,20,40,60)) {
    xx <- xmap(p)
    svg <- paste0(
      svg,
      '<line x1="',xx,'" y1="',top-11,
      '" x2="',xx,'" y2="',H-20,
      '" stroke="#eeeeee" stroke-width="1"/>',
      '<text x="',xx,'" y="',H-5,
      '" text-anchor="middle" font-size="9" fill="#999">',
      p, '%</text>'
    )
  }

  # metric rows
  for (r in seq_len(nrow(metrics))) {

    cy <- top + (r-1)*row_h

    svg <- paste0(
      svg,
      '<text x="8" y="',cy+4,
      '" font-size="11" font-weight="650" fill="#444">',
      metrics$label[r], '</text>',

      '<line x1="',left,'" y1="',cy,
      '" x2="',xmap(max_x),'" y2="',cy,
      '" stroke="#dddddd" stroke-width="2" stroke-linecap="round"/>'
    )

    vals <- m[[metrics$key[r]]]

    if (length(vals) > 1) {
      svg <- paste0(
        svg,
        '<line x1="',min(xmap(vals)),'" y1="',cy,
        '" x2="',max(xmap(vals)),'" y2="',cy,
        '" stroke="#bdbdbd" stroke-width="4" stroke-linecap="round" opacity=".55"/>'
      )
    }

    offsets <- if (length(vals)==2) c(-7,7) else c(-10,0,10)

    for (i in seq_along(vals)) {

      xx <- xmap(vals[i])
      yy <- cy + offsets[i]

      svg <- paste0(
        svg,
        '<circle cx="',xx,'" cy="',yy,
        '" r="6.5" fill="',group_palette[i],
        '" stroke="#fff" stroke-width="1.5"/>',

        '<text x="',xx+10,'" y="',yy+4,
        '" font-size="9.5" font-weight="700" fill="#555">',
        sprintf("%.1f%%",vals[i]), '</text>'
      )
    }
  }

  svg <- paste0(svg, '</svg>')
  HTML(svg)
}

  cats <- category_order[[variable]]

  div(
    class = "group-legend",

          class = "legend-item",
          span(
            class = "legend-dot",
            style = paste0("background:",group_palette[i],";")
          ),
          cats[i]
        )
      }
    )
  )
}

server <- function(input, output, session) {

  output$main_view <- renderUI({

    req(input$compare_by)

    cats <- category_order[[input$compare_by]]

    if (input$view == "cohort") {

      d <- state_df[state_df$variable == input$compare_by, ]

      cards <- lapply(
        cats,
        function(cat) {
          cohort_card(
            d[d$category == cat, ],
            cat
          )
        }
      )

      final <- d[d$year == 6, ]

      grad_vals <- final$graduated[match(cats, final$category)]
      out_vals <- final$outside[match(cats, final$category)]

      hi_grad <- cats[which.max(grad_vals)]
      lo_grad <- cats[which.min(grad_vals)]
      grad_gap <- max(grad_vals) - min(grad_vals)

      hi_out <- cats[which.max(out_vals)]
      lo_out <- cats[which.min(out_vals)]
      out_gap <- max(out_vals) - min(out_vals)

      tagList(

        div(
          class = "section-head",

          div(
            div(class="section-title","Cohort over time"),
            div(
              class="section-copy",
              "Each panel follows the same cohort through six annual observations. Each vertical bar is one annual observation and sums to 100%. The coloured segments show the distribution of educational states over time, following the visual logic of a TraMineR state distribution plot."
            )
          ),

          div(
            class="state-legend",
            state_legend("university","University"),
            state_legend("outside","Not in university"),
            state_legend("graduated","Graduated")
          )
        ),

        div(
          class="group-grid",
          style=paste0(
            "grid-template-columns:repeat(",
            length(cats),
            ",minmax(0,1fr));"
          ),
          cards
        ),

        div(
          class="reading",
          div(class="reading-label","At final observation"),
          div(
            class="reading-text",
            paste0(
              hi_grad, " has the highest graduation share (",
              sprintf("%.1f",max(grad_vals)), "%), compared with ",
              lo_grad, " (", sprintf("%.1f",min(grad_vals)),
              "%): a ", sprintf("%.1f",grad_gap),
              "-point gap. The largest share outside university is observed for ",
              hi_out, " (", sprintf("%.1f",max(out_vals)),
              "%), versus ", lo_out, " (", sprintf("%.1f",min(out_vals)),
              "%)."
            )
          )
        )
      )

    } else {

      m <- metrics_df[metrics_df$variable == input$compare_by, ]

      returned_vals <- m$returned[match(cats,m$category)]
      trans_vals <- m$transfer[match(cats,m$category)]
      comp_vals <- m$complex[match(cats,m$category)]

      tagList(

        div(
          class="section-head",

          div(
            div(class="section-title","Pathway dynamics"),
            div(
              class="section-copy",
              "Instead of looking only at the final outcome, this view compares events that occur during the trajectory: transfers, exits, returns and repeated state changes."
            )
          ),

          group_legend(input$compare_by)
        ),

        div(
          class="dynamics-card",
