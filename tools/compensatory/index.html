Compensatory Advantage Explorer

# ------------------------------------------------------------
# SERVER
# ------------------------------------------------------------

server <- function(input, output, session) {


  # Selected combination --------------------------------------

  selected <- reactive({

    dat[
      dat$performance == input$performance &
      dat$family == input$family,
    ]

  })


  # Numbers ----------------------------------------------------

  output$continue_value <- renderText({

    paste0(
      round(
        selected()$continuing * 100,
        1
      ),
      "%"
    )

  })


  output$dropout_value <- renderText({

    paste0(
      round(
        selected()$dropout * 100,
        1
      ),
      "%"
    )

  })


  output$transfer_value <- renderText({

    paste0(
      round(
        selected()$transfer * 100,
        1
      ),
      "%"
    )

  })


  # ----------------------------------------------------------
  # Dynamic Sankey-like flow
  # ----------------------------------------------------------

  output$flow <- renderUI({

    x <- selected()

    p1 <- x$continuing
    p2 <- x$dropout
    p3 <- x$transfer


    # Flow thickness

    width1 <- max(4, p1 * 60)
    width2 <- max(4, p2 * 60)
    width3 <- max(4, p3 * 60)


    html <- paste0(

      '<div style="
          width:100%;
          display:flex;
          justify-content:center;
          align-items:center;
        ">',


      '<svg
          viewBox="0 0 650 300"
          style="
            width:100%;
            max-width:620px;
            height:auto;
            display:block;
          ">',


      # --------------------------------------------------------
      # Title
      # --------------------------------------------------------

      '<text
          x="20"
          y="20"
          font-size="13"
          font-weight="600"
          fill="#222222">',

      input$family,
      ' family education · ',
      input$performance,
      '% performance',

      '</text>',


      # --------------------------------------------------------
      # Cohort box
      # --------------------------------------------------------

      '<rect
          x="25"
          y="112"
          width="125"
          height="76"
          rx="9"
          fill="#f5f5f5"
          stroke="#cccccc"/>',


      '<text
          x="87"
          y="137"
          text-anchor="middle"
          font-size="10"
          fill="#666666">
          SELECTED GROUP
       </text>',


      '<text
          x="87"
          y="164"
          text-anchor="middle"
          font-size="23"
          font-weight="600"
          fill="#222222">
          100
       </text>',


      '<text
          x="87"
          y="180"
          text-anchor="middle"
          font-size="10"
          fill="#666666">
          students
       </text>',


      # --------------------------------------------------------
      # Flows
      # --------------------------------------------------------

      '<path
          d="M150 140 C280 140, 330 63, 500 63"
          fill="none"
          stroke="#4E79A7"
          stroke-width="',
          width1,
          '"
          stroke-linecap="round"
          opacity="0.78"/>',


      '<path
          d="M150 150 C280 150, 330 150, 500 150"
          fill="none"
          stroke="#C95D5D"
          stroke-width="',
          width2,
          '"
          stroke-linecap="round"
          opacity="0.78"/>',


      '<path
          d="M150 160 C280 160, 330 237, 500 237"
          fill="none"
          stroke="#59A875"
          stroke-width="',
          width3,
          '"
          stroke-linecap="round"
          opacity="0.78"/>',


      # --------------------------------------------------------
      # Continue box
      # --------------------------------------------------------

      '<rect
          x="500"
          y="33"
          width="125"
          height="60"
          rx="9"
          fill="white"
          stroke="#cccccc"/>',


      '<text
          x="562"
          y="57"
          text-anchor="middle"
          font-size="12"
          fill="#444444">
          Continue
       </text>',


      '<text
          x="562"
          y="80"
          text-anchor="middle"
          font-size="20"
          font-weight="600"
          fill="#222222">',

      round(p1 * 100, 1),

      '%</text>',


      # --------------------------------------------------------
      # Dropout box
      # --------------------------------------------------------

      '<rect
          x="500"
          y="120"
          width="125"
          height="60"
          rx="9"
          fill="white"
          stroke="#cccccc"/>',


      '<text
          x="562"
          y="144"
          text-anchor="middle"
          font-size="12"
          fill="#444444">
          Drop out
       </text>',


      '<text
          x="562"
          y="167"
          text-anchor="middle"
          font-size="20"
          font-weight="600"
          fill="#222222">',

      round(p2 * 100, 1),

      '%</text>',


      # --------------------------------------------------------
      # Transfer box
      # --------------------------------------------------------

      '<rect
          x="500"
          y="207"
          width="125"
          height="60"
          rx="9"
          fill="white"
          stroke="#cccccc"/>',


      '<text
          x="562"
          y="231"
          text-anchor="middle"
          font-size="12"
          fill="#444444">
          Transfer
       </text>',


      '<text
          x="562"
          y="254"
          text-anchor="middle"
          font-size="20"
          font-weight="600"
          fill="#222222">',

      round(p3 * 100, 1),

      '%</text>',


      '</svg>',
      '</div>'

    )


    HTML(html)

  })


  # ----------------------------------------------------------
  # Comparison between family educational backgrounds
  # ----------------------------------------------------------

  output$comparison <- renderUI({

    temp <- dat[
      dat$performance == input$performance,
    ]


    rows <- lapply(

      seq_len(nrow(temp)),

      function(i) {

        r <- temp[i, ]


        tags$div(

          class = "compare-row",


          tags$div(

            r$family,

            class = "compare-label"

          ),


          tags$div(

            class = "compare-bar",


            tags$div(

              style = paste0(
                "width:",
                r$continuing * 100,
                "%;",
                "background:#4E79A7;"
              ),

              title = paste0(
                "Continue: ",
                round(
                  r$continuing * 100,
                  1
                ),
                "%"
              )

            ),


            tags$div(

              style = paste0(
                "width:",
                r$dropout * 100,
                "%;",
                "background:#C95D5D;"
              ),

              title = paste0(
                "Drop out: ",
                round(
                  r$dropout * 100,
                  1
                ),
                "%"
              )

            ),


            tags$div(

              style = paste0(
                "width:",
                r$transfer * 100,
                "%;",
                "background:#59A875;"
              ),

              title = paste0(
                "Transfer: ",
                round(
                  r$transfer * 100,
                  1
                ),
                "%"
              )

            )

          )

        )

      }

    )


    tagList(rows)

  })

}


# ------------------------------------------------------------
# RUN APP
# ------------------------------------------------------------

shinyApp(
  ui = ui,
  server = server
)
